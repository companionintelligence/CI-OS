== upgrade-login unit
# /usr/lib/systemd/user/gnome-initial-setup-upgrade-login.service
[Unit]
Description=GNOME Initial Setup

BindsTo=gnome-session.target
After=gnome-session.target

ConditionPathExists=%E/gnome-initial-setup-done
ConditionPathExists=!%E/gnome-initial-setup/upgrade-26.04-done

# Never run in GDM
ConditionUser=!@system
ConditionEnvironment=XDG_SESSION_CLASS=user

[Service]
Type=oneshot
ExecStart=/usr/libexec/gnome-initial-setup --upgrade-user
Restart=no
